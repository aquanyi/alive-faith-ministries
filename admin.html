1-Click Database Setup SQL

Copy this SQL script and paste it into the SQL Editor in your Supabase dashboard to create the tables, storage buckets, and security permissions:

-- 1. Create Devotions Table CREATE TABLE IF NOT EXISTS public.devotions ( id UUID DEFAULT gen_random_uuid() PRIMARY KEY, title TEXT NOT NULL, slug TEXT, scripture_ref TEXT NOT NULL, scripture_text TEXT NOT NULL, cover_image TEXT, content_html TEXT NOT NULL, prayer TEXT, confession TEXT, author TEXT DEFAULT 'Pastoral Ministry', category TEXT DEFAULT 'Grace & Faith', published_at DATE NOT NULL DEFAULT CURRENT_DATE, is_featured BOOLEAN DEFAULT false, created_at TIMESTAMP WITH TIME ZONE DEFAULT timezone('utc'::text, now()) NOT NULL ); -- 2. Enable Row Level Security (RLS) ALTER TABLE public.devotions ENABLE ROW LEVEL SECURITY; -- 3. Allow Public to Read Devotions CREATE POLICY "Public Read Devotions" ON public.devotions FOR SELECT USING (true); -- 4. Allow Authenticated Admins to Write & Edit CREATE POLICY "Admin Write Devotions" ON public.devotions FOR ALL TO authenticated USING (true) WITH CHECK (true); -- 5. Create Storage Bucket for Devotion Images INSERT INTO storage.buckets (id, name, public) VALUES ('devotion-images', 'devotion-images', true) ON CONFLICT (id) DO NOTHING; -- 6. Storage Bucket Security Policy CREATE POLICY "Public Read Devotion Images" ON storage.objects FOR SELECT USING (bucket_id = 'devotion-images'); CREATE POLICY "Admin Upload Devotion Images" ON storage.objects FOR INSERT TO authenticated WITH CHECK (bucket_id = 'devotion-images');
